% ── 3. Exp 1 method ───────────────────────────────────────────────────────────
\begin{frame}{Experiment 1 — Does the VIB Localize Attention?}
  \textbf{Inspired by Nagrani et al.\ (NeurIPS 2021)}, who showed attention
  bottlenecks concentrate LLM attention on sound-source regions.

  \medskip
  \textbf{Method}: same checkpoint, two conditions per clip.
  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{lll}
    \toprule
    Condition & Provider & What the LLM sees \\
    \midrule
    Without IB & \texttt{None} & raw encoder tokens \\
    With IB    & trained VIB  & compressed tokens \\
    \bottomrule
  \end{tabular}
  \end{center}
  Hook $q$-proj and $k$-proj in the last 8 decoder layers.
  Compute attention from the last prompt token over visual positions.
  \[\text{norm-entropy} = \frac{H(p)}{\log N}\!\in[0,1]
    \qquad
    \text{top-5\%-mass} = \!\!\sum_{k\in\text{top 5\%}}\!\! p_k\]
  Lower entropy and higher mass $\Rightarrow$ more localized.
\end{frame}

% ── 5. Exp 2 method ───────────────────────────────────────────────────────────
\begin{frame}{Experiment 2 — What Does the LLM Actually Attend To?}
  \textbf{Observation}: video gets $\sim$90\% of total attention mass.
  Is this because the model \emph{prefers} video, or just because there are
  more video tokens?

  \medskip
  \textbf{Method}: capture, per clip, per condition (raw / VIB):
  \begin{itemize}
    \item Token embeddings at layer-0 LLM input (video, audio, text positions)
    \item Full-sequence attention from last prompt token (all positions)
    \item Key norms at video / audio / text positions (last 8 layers)
  \end{itemize}
  Compute \textbf{per-token} attention = total mass $\div$ token count.

  \bigskip
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    & \textbf{Video} & \textbf{Audio} \\
    \midrule
    Token count & $\sim$2200 & $\sim$130 \\
    Fraction of sequence & 94\% & 6\% \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

% ── 6. Exp 2 results ──────────────────────────────────────────────────────────
\begin{frame}{Experiment 2 — Results}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Per-token attention (raw):}
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{lccc}
        \toprule
        & Video & Audio & Text \\
        \midrule
        Attn/token & $\approx$uniform & $1.5\times$ & $3\times$ \\
        Key norm   & 11--12           & 12--14       & \textbf{15.6} \\
        \bottomrule
      \end{tabular}

      \bigskip
      \textbf{VIB effect on audio:}
      \begin{tabular}{lc}
        \toprule
        Embedding norm ratio & $9.8\times$ \\
        Cosine similarity & \textcolor{Red}{0.24} \\
        $\Delta$ key norm after $k$-proj & \textcolor{Red}{$-2$} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Finding 1 — count, not preference}
        Per token, \textbf{audio gets more attention than video}.
        Video dominates total mass \emph{only} because of its
        17:1 token-count advantage.
      \end{block}

      \medskip
      \begin{alertblock}{Finding 2 — VIB suppresses audio keys}
        The VIB amplifies audio embeddings ($\times 9.8$) but
        \textbf{rotates them} (cos-sim $=0.24$).
        After $k$-proj, audio keys \emph{shrink} by $\sim2$,
        undermining audio's natural per-token advantage.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ── 7. Fixes ──────────────────────────────────────────────────────────────────
\begin{frame}{Two Targeted Fixes}
  \textbf{Fix 1 — Token-count equalization} (no retraining):
  \[z_j = \bigl[\,z_v \;\Big|\; \underbrace{\sqrt{\tfrac{n_v}{n_a}}}_{\approx\,4.1}
    \cdot z_a\,\bigr]\]
  Equalises total attention budget across modalities at inference time.

  \bigskip
  \textbf{Fix 2 — Direction-preservation loss} (added to training):
  \[\mathcal{L}_{\text{dir}} = \lambda\,
    \bigl(1 - \cos\!\operatorname{sim}(z_a,\, x_a^{\text{raw}})\bigr)
    \qquad \lambda = 0.1\]
  Penalises directional rotation of audio tokens.
  VIB can still compress (change norm) but must stay in the subspace
  that $k$-proj already responds to.

  \bigskip
  \textit{Both fixes are implemented in \texttt{AVModelBalanced} —
  training in progress, evaluation to follow.}
\end{frame}

% ── 8. Exp 3 ──────────────────────────────────────────────────────────────────
\begin{frame}{Experiment 3 — Prompt-Conditioned Modality Gate}
  \textbf{Idea}: ``What sound is being made?'' should weight audio more;
  ``What color is the car?'' should weight video more.

  \medskip
  \textbf{Architecture}: lightweight MLP on mean-pooled frozen prompt embeddings
  $\rightarrow$ softmax $\rightarrow$ probabilities $(p_v, p_a, p_t)$
  $\rightarrow$ scale injected tokens by $M \cdot p$.

  Zero-initialized output layer $\Rightarrow$ untrained gate = identity.

  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{lccc}
    \toprule
    \textbf{Prompt} & \textbf{$p_{\text{video}}$} & \textbf{$p_{\text{audio}}$} & \textbf{$p_{\text{text}}$} \\

    What color is the object? & & & \\
    Where is this scene?      & & & \\
    Is there music playing?   & & & \\
    \bottomrule

  \end{center}
