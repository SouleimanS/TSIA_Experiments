\begin{frame}{The problem in one picture}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      A multimodal LLM swallows \textbf{thousands of raw tokens} per clip:
      \begin{itemize}
        \item $\sim$2200 video patch tokens
        \item $\sim$130 audio tokens
        \item a handful of text tokens
      \end{itemize}
      \medskip
      Nothing tells the model \emph{which} of those tokens matter.

      \bigskip
      \textbf{Our idea:} insert a learnable \textcolor{Blue}{bottleneck} that
      \emph{compresses} each modality before it reaches the language model —
      and ask what changes.
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95,
          box/.style={rounded corners,draw=Ink,thick,minimum width=2.4cm,

\begin{frame}{What is a Variational Information Bottleneck?}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      A VIB squeezes each token stream through a noisy, low-capacity channel.
      It keeps what the task needs and throws away the rest.
      \[
        \mathcal{L} \;=\;
        \underbrace{\text{answer loss}}_{\text{be useful}}
        \;+\;
        \beta\underbrace{\big[\,\text{KL}_{\text{video}} + \text{KL}_{\text{audio}}\,\big]}_{\text{stay compressed}}
      \]
      \begin{itemize}
        \item Small $\beta$ $\Rightarrow$ light compression
        \item Large $\beta$ $\Rightarrow$ aggressive compression
        \item We train at $\beta = 7\times10^{-6}$, 2000 steps on AVQA
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.9]
        \fill[gray!15,rounded corners](0,0)rectangle(1.4,3);
        \node at (0.7,3.3){\small raw};
        \foreach \y in {0.2,0.5,...,2.8}\fill[Blue!50](0.2,\y)rectangle(1.2,\y+0.18);
        \draw[-{Stealth[length=2.5mm]},very thick,Orange](1.7,1.5)--(2.7,1.5)
          node[midway,above,font=\scriptsize]{VIB};
        \fill[Green!12,rounded corners](3,0.6)rectangle(4.4,2.4);
        \node at (3.7,2.7){\small compressed};
        \foreach \y in {0.9,1.4,1.9}\fill[Green!55](3.2,\y)rectangle(4.2,\y+0.22);
      \end{tikzpicture}
      \\[4pt]
      {\scriptsize Fewer effective dimensions \\ carry the signal forward.}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Same backbone, same prompt — the bottleneck is the \emph{only}
  thing we add. So any change we observe is caused by compression alone.}
\end{frame}

\section{Result 1}
\begin{frame}{Result 1 — The bottleneck focuses the model's attention}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      We read the LLM's attention over the video, \emph{with} and
      \emph{without} the bottleneck, from one checkpoint.

      \medskip
      Two simple measures of ``focus'':
      \begin{itemize}
        \item \textbf{Entropy} of the attention map — \textcolor{Green}{lower = sharper}
        \item \textbf{Top-5\% mass} — \textcolor{Green}{higher = sharper}
      \end{itemize}

      \medskip
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{lc}
        \toprule
        & with vs.\ without IB \\
        \midrule
        $\Delta$ entropy   & \textcolor{Green}{$-0.004$} \\
        $\Delta$ top-5\% mass & \textcolor{Green}{$+0.028$} \\
        \midrule
        clips improved & \textbf{30 / 30} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\textwidth]{0012y1s1bJI_000350_ib_compare.png}
      \\[2pt]
      {\scriptsize Left: frame.\quad Middle: without bottleneck.\quad
       Right: with bottleneck.}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{On \textbf{every one of 30 clips}, compression makes the model's
  visual attention more concentrated. A representation-level bottleneck
  reproduces the focusing effect previously seen only with attention-level
  bottlenecks (Nagrani et al., NeurIPS 2021).}
\end{frame}

\section{Result 2}
\begin{frame}{Result 2 — Why does video seem to ``win'' attention?}
  The LLM puts $\sim$90\% of its visual attention on video.
  Natural worry: \emph{is the model ignoring audio?}

  \medskip
  We split attention by modality and \textbf{normalize by token count}:
  \[
    \text{attention per token} \;=\;
    \frac{\text{total attention on a modality}}{\text{\# tokens in that modality}}
  \]

  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{lccc}
    \toprule
    & \textbf{Video} & \textbf{Audio} & \textbf{Text} \\
    \midrule
    \# tokens                 & $\sim$2200 & $\sim$130 & a few \\
    share of total attention  & $\sim$90\% & $\sim$6\%  & $\sim$4\% \\
    \rowcolor{Soft}
    \textbf{attention \emph{per token}} & baseline & \textcolor{Green}{$1.5\times$} & \textcolor{Green}{$3\times$} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  \takeaway{Video does \textbf{not} win on merit. Per token, audio and text
  actually get \emph{more} attention than video. Video only dominates the total
  because there are 17$\times$ more video tokens.}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Result 2 (cont.) — A hidden side-effect of compression}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      Looking closer at the audio tokens after the bottleneck:
      \medskip
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{lc}
        \toprule
        Audio embedding size & \textcolor{Blue}{$\times 9.8$ larger} \\
        Direction preserved? & \textcolor{Red}{cosine $=0.24$} \\
        Resulting ``key'' size & \textcolor{Red}{$-2$ (smaller)} \\
        \bottomrule
      \end{tabular}
      \medskip

      The bottleneck makes audio vectors \emph{bigger} but points them in a
      \textbf{new direction} the attention layer doesn't listen to — so audio's
      influence quietly drops.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95,>={Stealth[length=2.5mm]}]
        \draw[->,thick,Blue](0,0)--(2.6,1.4)node[right]{\small raw audio};
        \draw[->,thick,Orange,dashed](0,0)--(0.9,3.0)node[above]{\small after VIB};
        \draw[->,thick,Green](0,0)--(3.2,0.45)node[right]{\small what attention ``hears''};
        \draw[dotted,Blue](2.6,1.4)--(2.55,0.36);
        \draw[dotted,Orange](0.9,3.0)--(0.43,0.06);
        \node[Blue,font=\scriptsize,below]at(2.4,0.36){large};
        \node[Orange,font=\scriptsize,below right]at(0.5,0.06){small};
      \end{tikzpicture}
      \\[4pt]
      {\scriptsize Projection onto the ``listening'' direction\\shrinks even though the vector grew.}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Naive compression has an \textbf{unintended cost}: it rotates audio
  out of the subspace the LLM attends to. This is a concrete, measurable failure
  mode — and it points directly at the fix.}
\end{frame}

\section{Fix}
\begin{frame}{Two fixes that follow directly from the diagnosis}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Fix 1 — Balance the budget (free)}
        Scale audio tokens so the two modalities get \emph{equal} total
        attention budget:
        \[ z_a \;\leftarrow\; \sqrt{\tfrac{n_{\text{video}}}{n_{\text{audio}}}}\;z_a
           \;\approx\; 4.1\,z_a \]
        No retraining — pure inference-time rescaling.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Fix 2 — Keep audio pointing the right way}
        Add a small training penalty that stops the bottleneck rotating audio:
        \[ \mathcal{L}_{\text{dir}} = \lambda\big(1-\cos(z_a, z_a^{\text{raw}})\big) \]
        Compression of \emph{size} is still allowed; only harmful
        \emph{rotation} is penalized.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Both fixes are implemented and training is running.
  They turn an accidental bug (audio suppression) into a controllable design knob.}
\end{frame}

\section{Extension}
\begin{frame}{Extension — Let the question decide the modality mix}
  Different questions need different senses:
  \medskip
  \begin{center}
  \emph{``What \textcolor{Orange}{sound} is being made?''} $\rightarrow$ weight audio \qquad
  \emph{``What \textcolor{Blue}{color} is the car?''} $\rightarrow$ weight video
  \end{center}
  \medskip
  A tiny network reads the prompt and outputs a modality mix
  $(p_{\text{video}}, p_{\text{audio}}, p_{\text{text}})$ that rescales the
  tokens. It starts as the identity, so it can only help.

  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{lccc}
    \toprule
    \textbf{Prompt} & $p_{\text{video}}$ & $p_{\text{audio}}$ & $p_{\text{text}}$ \\
    \midrule
    What sound is being made? & \multicolumn{3}{c}{\textit{probe running — fills in live}} \\
    What color is the object?  & & & \\
    Is there music playing?    & & & \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  \takeaway{Gate is trained (final loss 0.36). The routing probe completes in
  minutes; numbers go straight into this table before the talk.}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Where this is going}
  \begin{enumerate}
    \item \textbf{Balanced model:} measure AVQA accuracy gain on audio-heavy
      questions once the fixes finish training.
    \item \textbf{Compression curve:} sweep $\beta$ and plot accuracy vs.\
      compression — find the sweet spot.
    \item \textbf{Perceive-then-reason:} let the model first \emph{name} the
      objects in a few frames, then condition compression on that description.
    \item \textbf{Modality routing:} confirm the gate sends audio questions to
      audio and visual questions to video.
  \end{enumerate}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item[\textcolor{Green}{\textbf{1}}] A representation-level bottleneck
      \textbf{sharpens} the LLM's visual attention — \textbf{30/30 clips}.
    \item[\textcolor{Green}{\textbf{2}}] Video dominates attention \textbf{by
      token count, not by merit}; per token, audio and text get more.
    \item[\textcolor{Green}{\textbf{3}}] Naive compression \textbf{accidentally
      mutes audio} by rotating it away from the attention subspace — measurable
      and fixable.
    \item[\textcolor{Green}{\textbf{4}}] Two principled fixes + a
      prompt-conditioned modality gate are implemented and training.
  \end{itemize}
  \vfill
  \begin{center}
    \Large\textcolor{Blue}{\textbf{Compression doesn't just shrink — it reshapes
    what the model pays attention to.}}
  \end{center}
\end{frame}
