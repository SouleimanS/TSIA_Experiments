\begin{frame}{How we measure ``reliance''}
  For each clip we record, from the last 8 layers of the LLM:
  \begin{enumerate}
    \item \textbf{Attention} from the answer position onto every input token,
      split into video / audio / text.
    \item \textbf{Per-token attention} = total attention on a modality
      $\div$ number of tokens in that modality. \\
      {\small This removes the unfair advantage of simply having more tokens.}
    \item \textbf{Key size} of each modality — bigger keys attract more attention
      inside the softmax.
  \end{enumerate}
  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{lcc}
    \toprule
    & \textbf{Video} & \textbf{Audio} \\
    \midrule
    number of tokens & $\sim$2200 & $\sim$130 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \takeaway{Right away, a clue: video has \textbf{17$\times$ more tokens} than
  audio. Could the 90\% just be a counting effect?}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\section{Answer, part 1}
\begin{frame}{Part 1 — It is the token count, not a preference}
  We normalize attention by the number of tokens in each modality:
  \begin{center}
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{lccc}
    \toprule
    & \textbf{Video} & \textbf{Audio} & \textbf{Text} \\
    \midrule
    share of total attention   & $\sim$90\% & $\sim$6\%  & $\sim$4\% \\
    number of tokens           & $\sim$2200 & $\sim$130 & a few \\
    \rowcolor{Soft}
    \textbf{attention \emph{per token}} & baseline & \textcolor{Green}{$\mathbf{1.5\times}$} & \textcolor{Green}{$\mathbf{3\times}$} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  Per token, \textbf{audio actually gets more attention than video} — and text
  gets the most of all. Video only dominates the \emph{total} because there are
  so many video tokens.
  \vfill
  \takeaway{The model does \textbf{not} prefer video. The 90\% is an arithmetic
  artefact of having 17$\times$ more video tokens. Audio is heard \emph{more}
  per token — it is simply outnumbered.}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\section{Answer, part 2}
\begin{frame}{Part 2 — Compression quietly makes it worse for audio}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      When we turn the bottleneck on and inspect the audio tokens:
      \medskip
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{lc}
        \toprule
        audio vector size & \textcolor{Blue}{$\times 9.8$ larger} \\
        direction kept?   & \textcolor{Red}{cosine $= 0.24$} \\
        resulting key size & \textcolor{Red}{$-2$ smaller} \\
        \bottomrule
      \end{tabular}
      \medskip

      The bottleneck makes audio vectors \emph{bigger} but turns them to point
      in a \textbf{new direction the attention layer doesn't listen to}.
      Net effect: audio's keys shrink and it is heard even less.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95,>={Stealth[length=2.5mm]}]
        \draw[->,thick,Blue](0,0)--(2.6,1.4)node[right]{\small raw audio};
        \draw[->,thick,Orange,dashed](0,0)--(0.9,3.0)node[above]{\small after VIB};
        \draw[->,thick,Green](0,0)--(3.2,0.45)node[right]{\small ``listening'' axis};
        \draw[dotted,Blue](2.6,1.4)--(2.55,0.36);
        \draw[dotted,Orange](0.9,3.0)--(0.43,0.06);
        \node[Blue,font=\scriptsize,below]at(2.4,0.36){large};
        \node[Orange,font=\scriptsize,below right]at(0.5,0.06){small};
      \end{tikzpicture}
      \\[4pt]
      {\scriptsize Bigger vector, but its overlap with the\\
      direction attention reads from \emph{shrinks}.}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Naive compression has an unintended cost: it rotates audio out of
  the subspace the LLM attends to. A measurable, fixable failure mode.}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{The full answer}
  \begin{center}
  \large \textbf{Why does the model rely on video more than audio?}
  \end{center}
  \medskip
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Cause 1 — Token-count imbalance}
        17$\times$ more video tokens than audio. Per token, audio is actually
        heard \emph{more}; it is just outnumbered in the total.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Cause 2 — Compression side-effect}
        The bottleneck rotates audio (cosine $0.24$) into a direction the
        attention layer ignores, shrinking its keys by $\sim$2 and suppressing
        it further.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    \textcolor{Blue}{\textbf{Video reliance is not a learned preference.
    It is structural (counting) plus an accidental compression artefact.}}
  \end{center}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{What the diagnosis tells us to do}
  Both causes suggest a direct fix:
  \medskip
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{For cause 1 — equalize the budget}
        Scale audio so the two modalities get equal total attention:
        \[ z_a \leftarrow \sqrt{\tfrac{n_{\text{video}}}{n_{\text{audio}}}}\,z_a
           \approx 4.1\,z_a \]
        No retraining needed.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{For cause 2 — keep audio aimed right}
        A small training penalty that forbids harmful rotation:
        \[ \lambda\big(1-\cos(z_a, z_a^{\text{raw}})\big) \]
        Compression of size is still allowed.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \takeaway{Both fixes are implemented and training now (details in backup).
  The diagnosis converts a vague worry into two concrete, testable knobs.}
\end{frame}

% ════════════════════════════════════════════════════════════════════════════
\begin{frame}{Summary}
  \textbf{Question:} why does the model rely on video more than audio?
  \medskip
  \begin{itemize}
    \item[\textcolor{Green}{\textbf{1}}] It is \textbf{not a preference.}
      Per token, audio (1.5$\times$) and text (3$\times$) get \emph{more}
      attention than video.
    \item[\textcolor{Green}{\textbf{2}}] Video dominates the total only because
      it has \textbf{17$\times$ more tokens}.
    \item[\textcolor{Green}{\textbf{3}}] Compression \textbf{makes it worse}:
      it rotates audio (cosine 0.24) out of the attention subspace, shrinking
      its keys.
    \item[\textcolor{Green}{\textbf{4}}] Two fixes follow directly — budget
      equalization and a direction-preserving loss.
  \end{itemize}
  \vfill
  \begin{center}
    \Large\textcolor{Blue}{\textbf{The model isn't deaf — audio is outnumbered,
    and compression turns down its volume.}}
  \end{center}
\end{frame}

\begin{frame}{\textsc{Backup} — The bottleneck sharpens visual attention}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      Same checkpoint, attention with vs.\ without the bottleneck.
      \medskip
      \renewcommand{\arraystretch}{1.4}

        \toprule

\begin{frame}{\textsc{Backup} — Prompt-conditioned modality gate}
  A tiny network reads the prompt and outputs a modality mix
  $(p_{\text{video}}, p_{\text{audio}}, p_{\text{text}})$ that rescales tokens.
  Zero-initialized $\Rightarrow$ starts as the identity.
  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{lccc}
    \toprule
    \textbf{Prompt} & $p_{\text{video}}$ & $p_{\text{audio}}$ & $p_{\text{text}}$ \\
    \midrule
    What sound is being made? & \multicolumn{3}{c}{\textit{probe pending}} \\
    What color is the object?  & & & \\
    Is there music playing?    & & & \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  Trained 2000 steps on AVQA, final loss 0.36. Routing probe runs in minutes.
\end{frame}

\begin{frame}{\textsc{Backup} — Roadmap}
  \begin{enumerate}
    \item \textbf{Balanced model:} AVQA accuracy on audio-heavy questions once
      the two fixes finish training.
    \item \textbf{Compression curve:} sweep $\beta$, plot accuracy vs.\
      compression.
    \item \textbf{Perceive-then-reason:} name objects in a few frames first,
      then condition compression on that description.
    \item \textbf{Routing check:} confirm the gate sends audio questions to
      audio and visual questions to video.
  \end{enumerate}
\end{frame}
