% 4 ───────────────────────────────────────────────────────────────────────────
\begin{frame}{How we measure attention (clean, causal)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      We use \textbf{gradient $\times$ input saliency}: how much each token
      changes the model's answer.
      \[ s_i = \mathrm{ReLU}\!\Big(\nabla_{x_i}\,\log p(\text{answer})\;\cdot\;x_i\Big) \]
      \begin{itemize}
        \item Causal — measures what drives the output
        \item Immune to attention sinks (unlike raw attention)
        \item Same checkpoint, bottleneck on vs.\ off
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{beamercolorbox}[sep=5pt,rounded=true]{}\colorbox{Soft}{\parbox{\dimexpr\textwidth-12pt}{%
      \textcolor{Blue}{\textbf{Why it matters.}} Raw decoder attention is
      dominated by sinks and looks like noise. Gradient saliency yields clean,
      object-aligned maps — and a fair with/without comparison.}}\end{beamercolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item[\textcolor{Green}{\textbf{1}}] \textbf{Where attention goes:} video
      dominates by token count (17:1), not preference; per token audio \& text
      get more.
    \item[\textcolor{Green}{\textbf{2}}] \textbf{The bottleneck localizes
      attention} onto the sound source — entropy $\downarrow$, top-mass
      $\uparrow$ (clean gradient maps).
    \item[\textcolor{Green}{\textbf{3}}] \textbf{Next:} balance the per-token
      budget and verify the projected gains on audio-heavy questions.
  \end{itemize}
  \vfill
  \begin{center}\Large\textcolor{Blue}{\textbf{A video bottleneck focuses the
  model on what matters in the frame.}}\end{center}
\end{frame}
